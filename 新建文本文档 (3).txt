<script>
(function(){
  const socket=io();
  const $=id=>document.getElementById(id);
  const playerId=localStorage.getItem('playerId')||(crypto.randomUUID?crypto.randomUUID():'p'+Math.random().toString(36).slice(2));
  localStorage.setItem('playerId',playerId);
  let state=null, selected=new Set(), toastTimer=null;

  function toast(m){
    const el=$('toast'); el.textContent=m; el.style.display='block';
    clearTimeout(toastTimer); toastTimer=setTimeout(()=>el.style.display='none',2200);
  }
  function name(){ return ($('name').value||'').trim()||'玩家'; }
  function esc(s){ return String(s).replace(/[&<>"']/g,m=>({'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;'}[m])); }

  function tryReconnect(){
    const rid=localStorage.getItem('roomId'); if(!rid) return;
    $('roomInput').value=rid;
    socket.emit('joinRoom',{roomId:rid,playerId,name:name()},r=>{
      if(!r||!r.ok) console.warn('重连失败',r&&r.error);
    });
  }
  socket.on('connect',tryReconnect);
  if(socket.connected) tryReconnect();

  socket.on('state',s=>{ state=s; localStorage.setItem('roomId',s.roomId); selected.clear(); render(); });

  function render(){
    if(!state) return;
    $('login').style.display='none'; $('game').style.display='flex';
    $('roomId').textContent=state.roomId;
    $('mult').textContent=state.bomb;
    $('status').textContent=state.status==='waiting'?'等待中':state.status==='playing'?'进行中':'已结束';
    const cur=state.status==='playing'&&state.players[state.cur];
    $('currentTxt').textContent=cur?'当前：'+cur.name:'';
    $('lastPlayTxt').textContent=state.last?'上家：'+state.last.label:'';

    $('players').innerHTML=state.players.map((p,i)=>{
      const tags=[];
      if(i===state.dealer) tags.push('<span class="tag dealer">庄</span>');
      if(p.id===state.yourId) tags.push('<span class="tag me">我</span>');
      if(!p.online) tags.push('<span class="tag offline">离线</span>');
      const lost=p.lost!=null?' · 本局输 '+p.lost:'';
      return '<div class="pcard '+(i===state.cur&&state.status==='playing'?'active':'')+'">'
        +'<div class="l1"><span>'+esc(p.name)+' '+tags.join('')+'</span><span>手牌 '+p.handCount+'</span></div>'
        +'<div class="l2"><span>出牌 '+p.playCnt+' 次</span><span>累计输分 '+(p.totalLost||0)+lost+'</span></div>'
        +'</div>';
    }).join('');

    $('hand').innerHTML=state.yourHand.map(c=>{
      const sel=selected.has(c.id)?'sel':'';
      const red=(c.suit==='♥'||c.suit==='♦')?'red':'';
      return '<div class="card '+red+' '+sel+'" data-id="'+c.id+'"><span>'+c.rank+'</span><span class="suit">'+(c.suit||'')+'</span></div>';
    }).join('');
    $('handCount').textContent=state.yourHand.length+' 张';

    document.querySelectorAll('.card').forEach(el=>{
      el.onclick=()=>{ const id=el.dataset.id; selected.has(id)?selected.delete(id):selected.add(id); render(); };
    });

    if(state.result){
      const r=state.result;
      $('result').style.display='block';
      $('result').innerHTML='<h3>'+esc(r.winnerName)+' 获胜 · 获得 '+r.gain+'</h3>'
        +'<p>公共倍数 '+r.bomb+' · 底分 '+state.bottom+'</p >'
        +'<ul>'+r.losers.map(l=>'<li>'+esc(l.name)+' 输 '+l.lost
          +'<br><span style="color:#94a3b8">剩余 '+l.remaining+' 张 × '+l.rm+(l.spring===2?' · 春天×2':'')+'</span></li>').join('')+'</ul>';
    } else $('result').style.display='none';

    const mi=state.players.findIndex(p=>p.id===state.yourId);
    const myTurn=state.status==='playing'&&state.cur===mi;
    const host=state.yourId===(state.players[0]&&state.players[0].id);
    $('hostBar').style.display=host?'flex':'none';
    $('start').style.display=state.status!=='playing'?'block':'none';
    $('reset').style.display=state.status==='playing'?'block':'none';
    $('play').disabled=!myTurn;
    $('pass').disabled=!myTurn||!state.last;
  }

  function emit(ev,d){ socket.emit(ev,d,r=>{ if(!r||!r.ok) toast((r&&r.error)||'操作失败'); }); }

  $('create').onclick=()=>socket.emit('createRoom',{playerId,name:name()},r=>{
    if(!r.ok) return toast(r.error);
    localStorage.setItem('roomId',r.roomId);
  });
  $('join').onclick=()=>{
    const rid=$('roomInput').value.trim(); if(!rid) return toast('输入房间码');
    socket.emit('joinRoom',{roomId:rid,playerId,name:name()},r=>{
      if(!r.ok) return toast(r.error);
      localStorage.setItem('roomId',rid);
    });
  };
  $('start').onclick=()=>state&&emit('startGame',{roomId:state.roomId,playerId});
  $('reset').onclick=()=>{ if(!state) return; if(!confirm('重置？累计输分保留'))return; emit('resetRoom',{roomId:state.roomId,playerId}); };
  $('play').onclick=()=>{ if(!state) return; if(!selected.size) return toast('选牌'); emit('play',{roomId:state.roomId,playerId,cardIds:[...selected]}); };
  $('pass').onclick=()=>state&&emit('pass',{roomId:state.roomId,playerId});
})();
</script>
